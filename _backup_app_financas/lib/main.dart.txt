void main() {
  WidgetsFlutterBinding.ensureInitialized();

  if (kIsWeb) {
    databaseFactory = createDatabaseFactoryFfiWeb(
      noWebWorker: true
    );
  } else {
    sqfliteFfiInit();
    databaseFactory = databaseFactoryFfi;
  }

  //if (defaultTargetPlatform == TargetPlatform.android) 

  runApp(const FinanceiroApp());
}

class FinanceiroApp extends StatelessWidget {
  const FinanceiroApp({super.key});

  @override
  Widget build(BuildContext context) {

    return BlocProvider(
      create: (context) => AuthCubit(repository: AuthRepository()),
      child: MaterialApp(
        debugShowCheckedModeBanner: false,
        title: 'Controle Financeiro',
        theme: ThemeData(
          colorScheme: ColorScheme.fromSeed(seedColor: Colors.teal)
        ),
        home: BlocBuilder<AuthCubit, AuthState>(
          builder: (context, state) {
            if (state is Authenticated) {
              final transacaoRepositoryApi = TransacaoRepository(accessToken: state.usuario.accessToken);
              final transacaoRepositoryDatabase = TransacaoLocalRepository();

              final syncService = SyncService(transacaoRepositoryApi, transacaoRepositoryDatabase);
              syncService.iniciarMonitoramento();
              
              return MultiBlocProvider(
                      providers: [
                        BlocProvider<TransacaoCubit>(
                          create: (context) => TransacaoCubit(
                            repositoryApi: transacaoRepositoryApi,
                            repositoryDatabase: transacaoRepositoryDatabase)..carregarTransacoes()
                        ),
                        BlocProvider<TransacaoFormCubit>(
                          create: (context) => TransacaoFormCubit(
                            transacaoCubit: context.read<TransacaoCubit>()
                          )
                        )
                      ],
                      child: const TelaPrincipal()
                );
            }

            return const LoginPage();
          }
        )
      ),
    );
